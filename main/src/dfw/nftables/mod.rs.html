24const NF_IP_PRI_NAT_DST: i16 = -100;
25const NF_IP_PRI_FILTER: i16 = 0;
26const NF_IP_PRI_NAT_SRC: i16 = 100;
27
28const NF_PRIORITY_IP_NAT_PREROUTING_DFW: i16 = NF_IP_PRI_NAT_DST - 5;
29const NF_PRIORITY_IP6_NAT_PREROUTING_DFW: i16 = NF_IP_PRI_NAT_DST - 5;
30const NF_PRIORITY_INET_FILTER_ANY_DFW: i16 = NF_IP_PRI_FILTER - 5;
31const NF_PRIORITY_IP_NAT_POSTROUTING_DFW: i16 = NF_IP_PRI_NAT_SRC - 5;
32const NF_PRIORITY_IP6_NAT_POSTROUTING_DFW: i16 = NF_IP_PRI_NAT_SRC - 5;
33
34const DFW_MARK: u32 = 0xdf;
35
36fn packet_mark_rules(
37    config: &crate::types::DFW<Nftables>,
38) -> impl Iterator<
39    Item = (
40        &crate::types::ContainerToWiderWorldRule<types::EgressRuleOptions>,
41        types::NftablesPacketMark,
42    ),
43> {
44    config
45        .container_to_wider_world
46        .iter()
47        .flat_map(|section| section.rules.iter().flatten())
48        .filter_map(|rule| Some((rule, rule.backend_options.as_ref()?.packet_mark?)))
49}
50
51fn egress_interface<'a>(
52    rule: &'a crate::types::ContainerToWiderWorldRule<types::EgressRuleOptions>,
53    primary: Option<&'a str>,
54) -> Option<&'a str> {
55    rule.external_network_interface.as_deref().or(primary)
56}
57
58fn is_concrete_interface(interface: &str) -> bool {
59    // Linux interface names are at most 15 bytes, excluding the terminating NUL.
60    !matches!(interface, "" | "." | "..")
61        && interface.len() <= 15
62        && interface
63            .bytes()
64            .all(|byte| byte.is_ascii_alphanumeric() || b"_.-".contains(&byte))
65}
66
67fn validate_packet_mark(mark: types::NftablesPacketMark) -> Result<()> {
68    if mark.preserve_mask & DFW_MARK != DFW_MARK {
69        return Err(DFWError::InvalidPacketMarkMask {
70            preserve_mask: mark.preserve_mask,
71            required_bits: DFW_MARK,
72        }
73        .into());
74    }
75    Ok(())
76}

78/// Marker struct to implement nftables as a firewall backend.
79#[derive(Debug)]
80pub struct Nftables;
81impl FirewallBackend for Nftables {
82    type Rule = String;
83    type Defaults = types::Defaults;
84    type EgressRuleOptions = types::EgressRuleOptions;
85
86    fn validate_config(config: &crate::types::DFW<Self>) -> Result<()> {
87        let primary = config
88            .global_defaults
89            .external_network_interfaces
90            .as_ref()
91            .and_then(|interfaces| interfaces.first())
92            .map(String::as_str);
93
94        for (rule, mark) in packet_mark_rules(config) {
95            validate_packet_mark(mark)?;
96            if !egress_interface(rule, primary).is_some_and(is_concrete_interface) {
97                return Err(DFWError::InvalidPacketMarkInterface.into());
98            }
99        }
100        Ok(())
101    }
102
103    fn apply(rules: Vec<Self::Rule>, ctx: &ProcessContext<Nftables>) -> Result<()> {
104        if ctx.dry_run {
105            info!(ctx.logger, "Performing dry-run, will not update any rules");
106        } else {
107            // To atomically update the ruleset, we need to write a file and pass that to `nft -f`.
108            let rule_file = tempfile::Builder::new().tempfile()?;
109            let rule_file_path = rule_file.as_ref().as_os_str().to_os_string();
110            debug!(ctx.logger, "Writing rules to temporary file";
111                   o!("file_path" => rule_file_path.to_string_lossy().into_owned()));
112            let mut writer = BufWriter::new(rule_file);
113
114            for rule in rules {
115                writeln!(writer, "{}", rule)?;
116            }
117            writer.flush()?;
118            trace!(ctx.logger, "Finished writing rules to temporary file");
119
120            info!(ctx.logger, "Applying rules (using nft)");
121            let output = Command::new("nft").arg("-f").arg(rule_file_path).output()?;
122            if !output.status.success() {
123                return Err(DFWError::NFTablesError {
124                    stdout: String::from_utf8_lossy(&output.stdout).into_owned(),
125                    stderr: String::from_utf8_lossy(&output.stderr).into_owned(),
126                }
127                .into());
128            } else {
129                return Ok(());
130            }
131        }
132
133        Ok(())
134    }
135}

190/// Representation of nftables chain hooks.
191///
192/// Order of hook execution:
193///
194/// * Ingress _(for netdev-family)_
195/// * Prerouting
196/// * Input
197/// * Forward
198/// * Output
199/// * Postrouting
200///
201/// ## Attribution
202///
203/// Parts of the documentation have been taken from
204/// <https://wiki.nftables.org/wiki-nftables/index.php/Configuring_chains>.
205#[derive(Debug, Clone, Copy, Display)]
206#[strum(serialize_all = "snake_case")]
207pub enum Hook {
208    /// Ingress allows traffic-filtering before pre-routing, after the packet traversed the NIC.
209    ///
210    /// Only available for the netdev-family.
211    Ingress,
212    /// Prerouting allows traffic-filtering before the packets have been routed.
213    Prerouting,
214    /// Input allows traffic-filtering for packets that have been routed to the local system.
215    Input,
216    /// Forward allows traffic-filtering for packets that were not routed to the local system.
217    Forward,
218    /// Output allows traffic-filtering for packets leaving the local system.
219    Output,
220    /// Postrouting allows traffic-filtering for already routed packets leaving the local system.
221    Postrouting,
222}
