49/// This trait is used to distinguish between different firewall backends.
50///
51/// To add a new firewall-backend create an empty struct implementing this trait.
52pub trait FirewallBackend: Sized
53where
54    DFW<Self>: Process<Self>,
55{
56    /// Associated type identifying the rule-type returned.
57    type Rule;
58    /// Associated type representing the firewall backend defaults/configuration.
59    type Defaults: Debug + DeserializeOwned;
60    /// Backend-specific options accepted by container-to-wider-world rules.
61    type EgressRuleOptions: Debug + DeserializeOwned;
62
63    /// Validate backend-specific configuration before discovering Docker state.
64    fn validate_config(_config: &DFW<Self>) -> Result<()> {
65        Ok(())
66    }
67
68    /// Apply the processed rules.
69    fn apply(rules: Vec<Self::Rule>, ctx: &ProcessContext<Self>) -> Result<()>;
70}
